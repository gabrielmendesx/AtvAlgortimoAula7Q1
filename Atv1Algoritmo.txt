#include <stdio.h>

int main() {
    float consumo, total = 0, media;
    int i;

    for (i = 1; i <= 5; i++) {
        printf("Consumo do morador %d (m3): ", i);
        scanf("%f", &consumo);

        if (consumo <= 20) {
            printf("Morador %d: dentro da media.\n", i);
        } else {
            printf("Morador %d: acima da media.\n", i);
        }

        total += consumo;
    }

    media = total / 5;
    printf("\nConsumo medio geral: %.2f m3\n", media);

    return 0;
}
